cmake_minimum_required(VERSION 3.16)
project(cpp_tsdb LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(tsdb src/codec.cpp src/engine.cpp)
target_include_directories(tsdb PUBLIC include)
add_executable(tsdb-cli src/main.cpp)
target_link_libraries(tsdb-cli PRIVATE tsdb)
add_executable(test_tsdb tests/test_tsdb.cpp)
target_link_libraries(test_tsdb PRIVATE tsdb)
if(NOT MSVC)
  target_compile_options(tsdb PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_tsdb COMMAND test_tsdb)
